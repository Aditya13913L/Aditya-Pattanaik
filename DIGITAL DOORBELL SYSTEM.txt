const int buttonPin = 2;
const int buzzerPin = 8;
const int LED_PIN = 13;

int visitorCount = 0;

int buttonState = HIGH;
int lastReading = HIGH;

unsigned long lastDebounceTime = 0;
const unsigned long debounceDelay = 50;

void playDoorbell()
{
  digitalWrite(LED_PIN, HIGH);
  
  tone(buzzerPin, 1000);
  delay(250);
  
  tone(buzzerPin, 1500);
  delay(350);
  
  tone(buzzerPin, 1000);
  delay(250);
  
  noTone(buzzerPin);
  
  digitalWrite(LED_PIN, LOW);
}

void setup() {
  pinMode(buttonPin, INPUT_PULLUP);
  pinMode(buzzerPin, OUTPUT);
  pinMode(LED_PIN, OUTPUT);

  Serial.begin(9600);

  Serial.println("Digital Doorbell System Ready!");
  Serial.println("Visitor Count: 0");
}

void loop() {
  int reading = digitalRead(buttonPin);

  // Detect a possible button state change
  if (reading != lastReading) {
    lastDebounceTime = millis();
  }

  // Accept the reading only after it stays stable
  if (millis() - lastDebounceTime >= debounceDelay) {
    if (reading != buttonState) {
      buttonState = reading;

      // Count only a new press, not the release
      if (buttonState == LOW) {
        visitorCount++;

        Serial.print("Doorbell Pressed! Visitor Count: ");
        Serial.println(visitorCount);

        playDoorbell();
      }
    }
  }

  lastReading = reading;
}
